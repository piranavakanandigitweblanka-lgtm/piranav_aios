4Fix Applied

One file changed — applies automatically to all 2,612+ pages.

File: shopify_projects/ledsone-uk-theme/snippets/seo-noindex.liquid — rendered on every page via theme.liquid line 148.

{%- comment -%} en-GB hreflang — Markets outputs 'en' (language-only);
this adds UK region signal piranav 2026-10-20 {%- endcomment -%}
{%- for locale in shop.published_locales -%}
  {%- if locale.iso_code == 'en' -%}
    {%- if request.locale.iso_code == 'en' -%}
      {%- assign en_canonical = canonical_url -%}
    {%- else -%}
      {%- assign en_canonical = canonical_url | replace: request.locale.root_url, '' -%}
    {%- endif -%}
    <link rel="alternate" hreflang="en-GB" href="{{ en_canonical }}">
    {%- break -%}
  {%- endif -%}
{%- endfor -%}
